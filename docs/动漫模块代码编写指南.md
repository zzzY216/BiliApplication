### ⚠️ 实现中发现的重要修正（写代码时务必以这些为准）

1. **所有模型必须显式 `@SerialName`**：项目用 kotlinx-serialization，**不会自动转换下划线**。缺少 `@SerialName("season_id")` 等注解，列表/详情解析会在运行时拿不到字段甚至崩溃。指南 §3 老代码块没写注解，已废弃，以 `PgcModels.kt` 为准。
2. **排行 `stat` 真实字段是 `follow`/`view`**（PiliPlus `pgc_rank/stat.dart`），不是 §3.1 里写的 views/danmakus 等。
3. **时间线分集真实字段**：`episodeId`/`pubIndex`/`pubTime(String)`/`follow`/`seasonId`/`cover`/`title`（PiliPlus `pgc_timeline/episode.dart`）。`pubTime` 是**字符串**，不是 Long——按 Long 解析会直接崩。
4. **`PagingState` 用 `anchorPosition`**（不是 `anchoredPosition`），参考 `BiliRecommendPagingSource`。
5. 排行接口需要 WBI 签名：已移植 PiliPlus `wbi_sign.dart` → `core/data/.../net/WbiSign.kt`（密钥来自 `/x/web-interface/nav` 的 `wbi_img`，按天缓存；hash 用按位拼接，不能用 `"%02x".format(Byte)`，负数会符号扩展出错）。

```kotlin
package com.software.core.data.repository

import androidx.paging.Pager
import androidx.paging.PagingConfig
import androidx.paging.PagingData
import androidx.paging.PagingSource
import androidx.paging.PagingState
import com.software.core.data.paging.PgcIndexPagingSource
import com.software.core.data.ResultExt.biliApiCall
import com.software.core.data.session.BiliSessionManager
import com.software.core.model.PlayUrlData
import com.software.core.model.pgc.*
import com.software.core.network.BiliApiNetwork
import com.software.core.network.PgcApiService
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.first
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class BangumiRepositoryImpl @Inject constructor(
    @BiliApiNetwork private val pgcApiService: PgcApiService,
    private val sessionManager: BiliSessionManager,
) : BangumiRepository {

    override fun getIndexPagingFlow(seasonType: Int): Flow<PagingData<PgcIndexItem>> =
        Pager(
            config = PagingConfig(pageSize = 21, enablePlaceholders = false),
            pagingSourceFactory = { PgcIndexPagingSource(pgcApiService, seasonType) }
        ).flow

    override suspend fun getRank(seasonType: Int, day: Int): Result<List<PgcRankItem>> =
        biliApiCall { pgcApiService.getRank(day = day, seasonType = seasonType) }
            .map { it.list.orEmpty() }

    override suspend fun getTimeline(types: Int): Result<List<TimelineResult>> =
        biliApiCall { pgcApiService.getTimeline(types = types) }
            .map { it.result.orEmpty() }

    override suspend fun getSeasonDetail(seasonId: Long): Result<SeasonDetail> =
        biliApiCall { pgcApiService.getSeasonDetail(seasonId) }
            .map { it.result ?: error("empty season detail") }

    override suspend fun getPlayUrl(epId: Long, cid: Long, qn: Int): Result<PlayUrlData> =
        biliApiCall { pgcApiService.getPlayUrl(epId, cid, qn) }
            .map { it.result?.videoInfo ?: error("empty playurl") }

    override suspend fun setFollow(seasonId: Long, follow: Boolean): Result<Unit> {
        val csrf = sessionManager.jctFlow.first()
        return biliApiCall {
            if (follow) pgcApiService.followSeason(seasonId, csrf)
            else pgcApiService.unfollowSeason(seasonId, csrf)
        }.map { }
    }

    override suspend fun getEpisodeCommunity(epId: Long): Result<PgcCommunityData> =
        biliApiCall { pgcApiService.getEpisodeCommunity(epId) }
}
```

**新建文件 3**：`core/data/src/main/java/com/software/core/data/paging/PgcIndexPagingSource.kt`

模板（仿照 `BiliRecommendPagingSource`）：

```kotlin
package com.software.core.data.paging

import androidx.paging.PagingSource
import androidx.paging.PagingState
import com.software.core.model.pgc.PgcIndexItem
import com.software.core.network.PgcApiService

class PgcIndexPagingSource(
    private val apiService: PgcApiService,
    private val seasonType: Int,
) : PagingSource<Int, PgcIndexItem>() {

    override suspend fun load(params: LoadParams<Int>): LoadResult<Int, PgcIndexItem> {
        return try {
            val page = params.key ?: 1
            val res = apiService.getIndexResult(
                page = page,
                pageSize = params.loadSize.coerceIn(1, 50),
                seasonType = seasonType,
            )
            val list = res.data?.list.orEmpty()
            if (res.code == 0) {
                LoadResult.Page(
                    data = list,
                    prevKey = if (page > 1) page - 1 else null,
                    nextKey = if (list.isNotEmpty()) page + 1 else null,
                )
            } else {
                LoadResult.Error(Exception(res.message))
            }
        } catch (e: Exception) {
            LoadResult.Error(e)
        }
    }

    override fun getRefreshKey(state: PagingState<Int, PgcIndexItem>): Int? =
        state.anchoredPosition?.let { anchor ->
            state.closestPageToPosition(anchor)?.prevKey?.plus(1)
                ?: state.closestPageToPosition(anchor)?.nextKey?.minus(1)
        }
}
```

```kotlin
@Composable
fun SeasonCard(
    cover: String?,          // 封面
    title: String,           // 标题
    badge: String?,          // 角标（"更新至XX话"）
    score: String?,          // 评分（可选）
    onClick: () -> Unit = {},
) {
    Column(
        modifier = Modifier
            .fillMaxWidth()
            .padding(horizontal = 12.dp, vertical = 6.dp)
            .clickable(onClick = onClick)
    ) {
        AsyncImage(
            model = cover,
            contentDescription = title,
            modifier = Modifier.fillMaxWidth().aspectRatio(3f / 4f),  // 番剧封面是竖版 3:4，别用 16:9
            contentScale = ContentScale.Crop,
        )
        Spacer(Modifier.height(6.dp))
        Text(title, style = MaterialTheme.typography.titleSmall, maxLines = 2, overflow = TextOverflow.Ellipsis)
        if (badge != null || score != null) {
            Spacer(Modifier.height(4.dp))
            Text(
                text = listOfNotNull(badge, score).joinToString(" · "),
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
                maxLines = 1,
            )
        }
    }
}

// 两个重载：索引项 / 排行项 → 映射到 SeasonCard 参数
@Composable
fun SeasonCard(item: PgcIndexItem, onClick: () -> Unit = {}) =

@Composable

    SeasonCard(item.cover, item.title.orEmpty(), item.newEp?.desc, item.stat?.views?.let { "播放 $it" }, onClick)
```

  进度续播用 `result.lastPlayTime`。**duration 单位是毫秒**。
- **第 13 步（互动）**：追番按钮（`setFollow`）、`getEpisodeCommunity` 显示赞/币/藏、三连、短评（`pgc/review/*`，见《API 接口文档》§11）。

## 常见错误速查

| 报错 | 原因 | 修法 |
| --- | --- | --- |
| `404` 请求失败 | Retrofit 路径带前导 `/` | 去掉 `/`：`"x/pgc/..."` |
| `-403` 排行接口 | 需要 WBI 签名 | 暂时用索引代替排行；或实现 WbiSign（密钥来自 `/x/web-interface/nav` 的 `wbi_img`） |
| `BiliResponse 解析失败 / JsonDecodingException` | 模型字段名/类型和响应不一致 | 对照《开发指南》§3 的字段逐个检查 |
| 分页重复请求第一页 | PagingSource `nextKey` 逻辑错 | 按第 4 步模板写 |
| `csrf` 无效 / `-111` | csrf 空 | 确认 `sessionManager.jctFlow.first()` 能读到 bili_jct（登录后才有） |
| 首页点了 Tab 没反应 | MusicTopBar 状态没提升 | 第 6 步没做 |
